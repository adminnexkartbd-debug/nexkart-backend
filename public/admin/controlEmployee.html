Database setup SQL · defaultdb

Run once in DBeaver before posting banners.

USE `defaultdb`;
CREATE TABLE IF NOT EXISTS `offer_banners` (
  `id` BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  `title` VARCHAR(150) NOT NULL,
  `alt_text` VARCHAR(200) NOT NULL DEFAULT '',
  `image_url` VARCHAR(1000) NOT NULL,
  `image_public_id` VARCHAR(255) NOT NULL,
  `target_url` VARCHAR(1000) NOT NULL DEFAULT '',
  `sort_order` INT UNSIGNED NOT NULL DEFAULT 0,
  `is_active` TINYINT(1) NOT NULL DEFAULT 1,
  `version` INT UNSIGNED NOT NULL DEFAULT 1,
  `created_at` TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
  `updated_at` TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  KEY `idx_offer_banners_active_order` (`is_active`,`sort_order`,`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;
